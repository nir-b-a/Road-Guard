# =============================================================================
# Road Guard — Python pipeline requirements
# Target: Windows 10/11, NVIDIA GPU (CUDA 11.8), Python 3.10
# Versions below are the EXACT ones from the working .venv (proven to run
# main.py + the violation harness on yolo11x + torch 2.1.2+cu118).
# =============================================================================
#
# IMPORTANT — DO NOT `pip install -r requirements.txt` blindly.
# torch/torchvision are NOT listed here: they must come from the CUDA-specific
# wheel index, or pip pulls a CPU-only build and GPU inference silently breaks.
# Install in this order:
#
#   STEP 1  Create + activate a clean Python 3.10 env
#             python -m venv .venv
#             .venv\Scripts\activate            (PowerShell: .venv\Scripts\Activate.ps1)
#           (or:  conda create -n roadguard python=3.10 -y && conda activate roadguard)
#
#   STEP 2  Install torch built for CUDA 11.8 (Pascal-compatible, e.g. GTX 1060):
#             pip install torch==2.1.2 torchvision==0.16.2 \
#                 --index-url https://download.pytorch.org/whl/cu118
#           (CPU-only fallback, no GPU:  pip install torch==2.1.2 torchvision==0.16.2)
#
#   STEP 3  Install everything in this file:
#             pip install -r requirements.txt
#
# The YOLO base weights are auto-downloaded by ultralytics on first run
# (yolo11x.pt ~109 MB, yolov8m.pt for the violation harness). They are gitignored.
# -----------------------------------------------------------------------------

# --- CORE: required to run main.py (speed estimation) -------------------------
# numpy<2 — numpy 2.0 broke the ABI for many compiled CV/ML wheels. Keep pinned.
numpy==1.26.4
# YOLO11 detection + tracking. 8.4.x supports yolo11x and works with torch 2.1.x.
ultralytics==8.4.68
# OpenCV — headed build (keeps cv2.imshow available for the debug windows).
opencv-python==4.10.0.84
# Linear-assignment backend for the BoT-SORT / ByteTrack trackers (ultralytics).

# Savitzky-Golay smoothing + Kalman/RTS math (speed_estimation/), plus the
# least-squares camera-height solve and 3D rotations in ground_distance/.
scipy==1.13.1
# Per-vehicle speed/distance plots + ego-speed plot (speed_estimation/draw_vehicle_plots.py).
matplotlib==3.10.9
# Overpass speed-limit lookup for the overspeed step (speed_estimation/speed_limit_lookup.py),
# and the GPU worker's HTTP calls to the job server (worker.py).
requests==2.28.1
# Reads the dashcam sensor CSVs in ground_distance/sensors.py (also pulled in by ultralytics).
pandas==2.2.2
pillow==10.4.0
pyyaml==6.0.2
tqdm==4.66.5

# --- GPU WORKER: only needed to run the cloud worker (worker.py) --------------

# pull each job's files from / push evidence clips to Cloudflare R2; botocore and
# s3transfer install automatically alongside it.
boto3==1.35.36

# --- OPTIONAL: license-plate recognition (lpr/) -------------------------------

# --- OPTIONAL: dataset download (tests_videos/download_videos.py) -------------
yt-dlp
